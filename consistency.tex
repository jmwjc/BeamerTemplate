\begin{frame}{薄壳问题的Hu-Washizu变分原理伽辽金弱形式}
    \textbf{薄壳薄膜应变部分变分一致性条件}
    \begin{equation*}
        \int_\Omega \underbrace{\delta \varepsilon_{\alpha\beta} N^{\alpha\beta}}_{\displaystyle \frac{1}{2}({\color{red}\boldsymbol a_{\alpha} N^{\alpha\beta}} \cdot {\color{blue}\delta \boldsymbol u_{,\beta}} + {\color{blue}\delta \boldsymbol u_{,\alpha}} \cdot {\color{red}\boldsymbol a_\beta N^{\alpha\beta}})} d\Omega = \int_\Gamma \delta \boldsymbol u \cdot \underbrace{\boldsymbol T_N}_{\displaystyle {\color{red} \boldsymbol a_\alpha N^{\alpha\beta}} n_\beta} d\Gamma - \int_\Omega \delta \boldsymbol u \cdot \underbrace{\boldsymbol b_N}_{\displaystyle ({\color{red}\boldsymbol a_\alpha N^{\alpha\beta}})\vert_\beta} d\Omega
    \end{equation*}

    \vspace{-10pt}

    \begin{itemize}
    \item 假定应变:
        $\displaystyle \boldsymbol u_{,\alpha}(\boldsymbol \xi) = \boldsymbol q^T(\boldsymbol \xi) \boldsymbol d_\alpha^{\varepsilon}
        \qquad \quad \! \Rightarrow \quad
        \varepsilon_{\alpha\beta}(\boldsymbol \xi) = \boldsymbol q^T(\boldsymbol \xi)\frac{1}{2}(\boldsymbol a_\alpha \cdot \boldsymbol d_\beta^\varepsilon + \boldsymbol a_\beta \cdot \boldsymbol d^\varepsilon_\alpha)$
    \item 假定应力:
        $\displaystyle \boldsymbol a_\alpha N^{\alpha\beta}(\boldsymbol \xi) = \boldsymbol q^T(\boldsymbol \xi) \boldsymbol d_N^\beta
        \quad\Rightarrow \quad
        N^{\alpha\beta}(\boldsymbol \xi) = \boldsymbol q^T(\boldsymbol \xi) \boldsymbol a^\alpha \cdot \boldsymbol d_N^\beta$
    \end{itemize}
    \textbf{薄膜应变部分Hu-Washizu变分原理}
    {\footnotesize
    \begin{equation*}
        - \int_\Omega h \delta \varepsilon_{\alpha\beta} C^{\alpha\beta\gamma\eta}\varepsilon_{\gamma\eta}d\Omega 
        + \int_\Omega \delta \varepsilon_{\alpha\beta} N^{\alpha\beta} d\Omega = 0
    \end{equation*}
    \begin{equation*}
        \int_\Omega \delta N^{\alpha\beta} \varepsilon_{\alpha\beta} d\Omega
        - \int_\Gamma \delta \boldsymbol T_N \cdot \boldsymbol u d\Gamma 
        + \int_\Omega \delta \boldsymbol b_N \cdot \boldsymbol v d\Omega 
        + \int_{\Gamma_u} \delta \boldsymbol T_N \cdot \boldsymbol u d\Gamma 
        = \int_{\Gamma_u} \delta \boldsymbol T_N \cdot \bar{\boldsymbol u} d\Gamma 
    \end{equation*}}
\end{frame}

\begin{frame}{与Nitsche法对比}

    \textbf{一致项}

    \alert{\bfseries$\checkmark$形函数二阶及以上导数采用光滑梯度代替，提高计算效率}

    \vspace{10pt}

    \begin{columns}
        \column{0.45\textwidth}

        \textrm{HW法:}

        \vspace{-10pt}

        {\footnotesize
        \begin{equation*}
            \begin{aligned}
            \tilde{\boldsymbol K}_{IJ} &= - \int_{\Gamma_v} (\Psi_I \alert{\tilde{\boldsymbol T}_{NJ}(\Psi_{J,\alpha})} + \alert{\tilde{\boldsymbol T}_{NI}(\Psi_{I,\alpha})} \Psi_J) d\Gamma \\
            &+ \int_{\Gamma_\theta} (\Psi_{I,\gamma} n^\gamma \boldsymbol a_3 \tilde{\boldsymbol M}_{\boldsymbol{nn}J} + \boldsymbol a_3 \tilde{\boldsymbol M}_{\boldsymbol{nn}I} \Psi_{J,\gamma}n^\gamma)d\Gamma \\
            &+ ([[\Psi_I \boldsymbol a_3 \tilde{\boldsymbol P}_J]] + [[\tilde{\boldsymbol P}_I \boldsymbol a_3 \Psi_J]])_{\boldsymbol x \in C_v} 
            \end{aligned}
        \end{equation*}}
        \column{0.45\textwidth}

        \textrm{Nitsche法:}

        \vspace{-10pt}

        {\footnotesize
        \begin{equation*}
            \begin{aligned}
                \boldsymbol K^c_{IJ} = &- \int_{\Gamma_v} (\Psi_I \alert{\boldsymbol T_{NJ}(\Psi_{J,\alpha\beta\gamma})} + \alert{\boldsymbol T_{NI}(\Psi_{I,\alpha\beta\gamma})} \Psi_J) d\Gamma \\
                &+ \int_{\Gamma_\theta} (\Psi_{I,\gamma} n^\gamma \boldsymbol a_3 \boldsymbol M_{\boldsymbol{nn}J} + \boldsymbol a_3 \boldsymbol M_{\boldsymbol{nn}I} \Psi_{J,\gamma}n^\gamma)d\Gamma \\
                & + ([[\Psi_I \boldsymbol a_3 \boldsymbol P_J]] + [[\boldsymbol P_I \boldsymbol a_3 \Psi_J]])_{\boldsymbol x \in C_v}
            \end{aligned}
        \end{equation*}}
    \end{columns}

    \vspace{10pt}
